## 4.4 权限管理中心

### 权限模型设计

```
RBAC + ABAC 混合模型：

用户 ──has──> 角色 ──has──> 权限

例：
用户 "张三" 
  has 角色 "销售员" ──has──> 权限 "创建订单"
  has 角色 "销售员" ──has──> 权限 "编辑自己的订单"

条件表达式 (ABAC)：
  用户可编辑订单 IF:
    - 用户角色 = "销售员" 
    AND 订单创建人 = 当前用户
    AND 订单状态 ≠ "已完成"
```

## 4.8 数据库设计初稿

### 核心表结构

```sql
-- 应用管理
CREATE TABLE applications (
  id UUID PRIMARY KEY,
  tenant_id UUID NOT NULL,
  name VARCHAR(255) NOT NULL,
  description TEXT,
  app_type VARCHAR(50),
  created_by UUID,
  created_at TIMESTAMP,
  updated_at TIMESTAMP,
  status VARCHAR(20), -- draft, production, archived
  version VARCHAR(50)
);

-- 表单元数据
CREATE TABLE forms (
  id UUID PRIMARY KEY,
  app_id UUID REFERENCES applications(id),
  name VARCHAR(255),
  description TEXT,
  form_schema JSONB, -- 存储字段定义
  created_at TIMESTAMP,
  updated_at TIMESTAMP
);

-- 表单字段
CREATE TABLE form_fields (
  id UUID PRIMARY KEY,
  form_id UUID REFERENCES forms(id),
  field_name VARCHAR(255),
  field_type VARCHAR(50), -- text, number, date, etc.
  is_required BOOLEAN,
  validation_rules JSONB, -- 验证规则
  permissions JSONB, -- 字段级权限
  sort_order INTEGER
);

-- 工作流定义
CREATE TABLE workflows (
  id UUID PRIMARY KEY,
  app_id UUID REFERENCES applications(id),
  name VARCHAR(255),
  bpmn_xml TEXT, -- BPMN 2.0 XML
  created_at TIMESTAMP,
  updated_at TIMESTAMP,
  is_active BOOLEAN,
  version VARCHAR(50)
);

-- 权限配置
CREATE TABLE permissions (
  id UUID PRIMARY KEY,
  role_id UUID,
  resource VARCHAR(255),
  action VARCHAR(255),
  conditions JSONB -- ABAC 条件表达式
);

-- 审计日志
CREATE TABLE audit_logs (
  id UUID PRIMARY KEY,
  tenant_id UUID,
  user_id UUID,
  action VARCHAR(255),
  resource_type VARCHAR(50),
  resource_id UUID,
  changes JSONB,
  created_at TIMESTAMP,
  ip_address VARCHAR(50),
  user_agent TEXT
);

-- 应用数据 (动态表)
-- 每个表单一个动态表，表名: tenant_{tenantId}_form_{formId}
-- 例: tenant_001_form_123

-- 用户
CREATE TABLE users (
  id UUID PRIMARY KEY,
  tenant_id UUID,
  email VARCHAR(255) UNIQUE,
  password_hash VARCHAR(255),
  full_name VARCHAR(255),
  created_at TIMESTAMP,
  updated_at TIMESTAMP
);

-- 角色
CREATE TABLE roles (
  id UUID PRIMARY KEY,
  tenant_id UUID,
  name VARCHAR(255),
  description TEXT,
  created_at TIMESTAMP
);

-- 用户角色关联
CREATE TABLE user_roles (
  user_id UUID REFERENCES users(id),
  role_id UUID REFERENCES roles(id),
  PRIMARY KEY (user_id, role_id)
);
```
